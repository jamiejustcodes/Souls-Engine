cmake_minimum_required(VERSION 3.28)
project(SoulsEngine VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SOULS_BUILD_SHELLS "Build SDL runtime and editor" ON)
option(SOULS_ASAN "Enable AddressSanitizer in Debug" OFF)
option(SOULS_UBSAN "Enable UndefinedBehaviorSanitizer in Debug (GCC/Clang)" OFF)
option(SOULS_WARNINGS_AS_ERRORS "Treat engine warnings as errors" ON)
option(SOULS_BUILD_BENCHMARKS "Build deterministic core microbenchmark" ON)
option(SOULS_VULKAN_COMPILE_CHECK "Build Vulkan source, shader and runtime checks on Windows" OFF)
include(CTest)

add_library(SoulsBuildOptions INTERFACE)
target_compile_features(SoulsBuildOptions INTERFACE cxx_std_23)
if(MSVC)
  string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
  target_compile_options(SoulsBuildOptions INTERFACE /W4 /permissive- /EHs-c- /utf-8 /Zc:__cplusplus)
  target_compile_definitions(SoulsBuildOptions INTERFACE _HAS_EXCEPTIONS=0 NOMINMAX WIN32_LEAN_AND_MEAN)
  if(SOULS_WARNINGS_AS_ERRORS)
    target_compile_options(SoulsBuildOptions INTERFACE /WX)
  endif()
  if(SOULS_ASAN)
    string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
    string(REPLACE "/RTC1" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
    target_compile_options(SoulsBuildOptions INTERFACE $<$<CONFIG:Debug>:/fsanitize=address> $<$<CONFIG:Debug>:/Zi>)
    target_link_options(SoulsBuildOptions INTERFACE $<$<CONFIG:Debug>:/INCREMENTAL:NO>)
  endif()
  if(SOULS_UBSAN)
    message(FATAL_ERROR "MSVC does not provide UBSan; use the Linux sanitize preset")
  endif()
else()
  target_compile_options(SoulsBuildOptions INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -fno-exceptions)
  if(SOULS_WARNINGS_AS_ERRORS)
    target_compile_options(SoulsBuildOptions INTERFACE -Werror)
  endif()
  foreach(sanitizer IN ITEMS ASAN UBSAN)
    if(SOULS_${sanitizer})
      if(sanitizer STREQUAL "ASAN")
        set(flag -fsanitize=address)
      else()
        set(flag -fsanitize=undefined)
      endif()
      target_compile_options(SoulsBuildOptions INTERFACE $<$<CONFIG:Debug>:${flag}> $<$<CONFIG:Debug>:-fno-omit-frame-pointer>)
      target_link_options(SoulsBuildOptions INTERFACE $<$<CONFIG:Debug>:${flag}>)
    endif()
  endforeach()
endif()

add_library(SoulsCore STATIC src/core/Core.cpp)
target_include_directories(SoulsCore PUBLIC include)
target_link_libraries(SoulsCore PUBLIC SoulsBuildOptions)
if(BUILD_TESTING)
  add_executable(SoulsCoreTests tests/CoreTests.cpp)
  target_link_libraries(SoulsCoreTests PRIVATE SoulsCore)
  add_test(NAME core.contracts COMMAND SoulsCoreTests)
endif()
if(SOULS_BUILD_BENCHMARKS)
  add_executable(SoulsCoreBench benchmarks/CoreBench.cpp)
  target_link_libraries(SoulsCoreBench PRIVATE SoulsCore)
endif()

if(SOULS_BUILD_SHELLS)
  include(cmake/Dependencies.cmake)
  if(WIN32)
    set(SOULS_BACKEND D3D12)
    set(rhi_source src/rhi/D3D12.cpp)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(SOULS_BACKEND VULKAN)
    set(rhi_source src/rhi/Vulkan.cpp)
  else()
    message(FATAL_ERROR "Souls Engine supports Windows (D3D12) and Linux (Vulkan) only")
  endif()
  add_library(SoulsRHI STATIC ${rhi_source})
  target_link_libraries(SoulsRHI PUBLIC SoulsCore PRIVATE SDL3::SDL3)
  target_compile_definitions(SoulsRHI PUBLIC SOULS_RHI_${SOULS_BACKEND}=1)
  target_include_directories(SoulsRHI PRIVATE src)
  if(WIN32)
    target_include_directories(SoulsRHI SYSTEM PUBLIC "${agility_SOURCE_DIR}/build/native/include")
    target_link_libraries(SoulsRHI PRIVATE d3d12 dxgi dxguid)
    if(SOULS_VULKAN_COMPILE_CHECK)
      add_library(SoulsVulkanCompileCheck STATIC src/rhi/Vulkan.cpp)
      target_include_directories(SoulsVulkanCompileCheck PRIVATE include src)
      target_compile_definitions(SoulsVulkanCompileCheck PRIVATE SOULS_RHI_VULKAN=1 VK_NO_PROTOTYPES)
      target_link_libraries(SoulsVulkanCompileCheck PRIVATE SoulsCore SDL3::SDL3 volk)
      add_library(SoulsVulkanUICompileCheck OBJECT src/editor/GraphicsUI.cpp src/editor/main.cpp)
      target_include_directories(SoulsVulkanUICompileCheck PRIVATE include src)
      target_compile_definitions(SoulsVulkanUICompileCheck PRIVATE SOULS_RHI_VULKAN=1 IMGUI_IMPL_VULKAN_USE_VOLK VK_NO_PROTOTYPES)
      target_link_libraries(SoulsVulkanUICompileCheck PRIVATE SoulsCore SDL3::SDL3 volk SoulsImGui SoulsImPlot)
    endif()
  else()
    target_link_libraries(SoulsRHI PRIVATE volk)
  endif()
  add_library(SoulsRenderer STATIC src/renderer/Renderer.cpp)
  target_link_libraries(SoulsRenderer PUBLIC SoulsRHI)
  add_library(SoulsScene STATIC src/scene/Scene.cpp)
  target_include_directories(SoulsScene SYSTEM PRIVATE "${flecs_SOURCE_DIR}/include")
  target_link_libraries(SoulsScene PUBLIC SoulsCore flecs::flecs_static)
  target_link_libraries(SoulsRenderer PUBLIC SoulsScene)
  include(cmake/Shaders.cmake)
  add_executable(SoulsRuntime src/runtime/main.cpp)
  target_link_libraries(SoulsRuntime PRIVATE SoulsRenderer SoulsScene SDL3::SDL3)
  add_executable(SoulsEditor src/editor/main.cpp src/editor/GraphicsUI.cpp src/editor/Workspace.cpp src/editor/WorkspaceActors.cpp)
  target_include_directories(SoulsEditor PRIVATE src)
  target_link_libraries(SoulsEditor PRIVATE SoulsRenderer SoulsScene SoulsImGui SoulsImPlot SDL3::SDL3)
  if(NOT WIN32)
    target_link_libraries(SoulsEditor PRIVATE volk)
  endif()
  foreach(app IN ITEMS SoulsRuntime SoulsEditor)
    if(WIN32)
      target_sources(${app} PRIVATE src/platform/Agility.cpp)
      target_compile_definitions(${app} PRIVATE SOULS_AGILITY_VERSION=616)
      add_custom_command(TARGET ${app} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${app}>/D3D12"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${agility_SOURCE_DIR}/build/native/bin/x64/D3D12Core.dll" "$<TARGET_FILE_DIR:${app}>/D3D12"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${agility_SOURCE_DIR}/build/native/bin/x64/d3d12SDKLayers.dll" "$<TARGET_FILE_DIR:${app}>/D3D12")
    endif()
    if(BUILD_TESTING)
      add_test(NAME ${app}.smoke COMMAND ${app} --smoke 120)
      set_tests_properties(${app}.smoke PROPERTIES LABELS "gpu;interactive" TIMEOUT 90)
    endif()
  endforeach()
  if(BUILD_TESTING)
    add_executable(SoulsSceneTests tests/SceneTests.cpp)
    target_link_libraries(SoulsSceneTests PRIVATE SoulsScene)
    add_test(NAME scene.contracts COMMAND SoulsSceneTests)
    add_executable(SoulsAllocationTests tests/AllocationTests.cpp)
    target_include_directories(SoulsAllocationTests SYSTEM PRIVATE "${flecs_SOURCE_DIR}/include")
    target_link_libraries(SoulsAllocationTests PRIVATE SoulsScene)
    add_test(NAME scene.frame_allocations COMMAND SoulsAllocationTests)
  endif()
endif()

# ASan executables must be runnable outside a Visual Studio developer shell.
if(MSVC AND SOULS_ASAN)
  get_filename_component(souls_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
  find_file(SOULS_ASAN_RUNTIME_RELEASE clang_rt.asan_dynamic-x86_64.dll PATHS "${souls_compiler_bin}" NO_DEFAULT_PATH REQUIRED)
  find_file(SOULS_ASAN_RUNTIME_DEBUG clang_rt.asan_dbg_dynamic-x86_64.dll PATHS "${souls_compiler_bin}" NO_DEFAULT_PATH)
  foreach(app IN ITEMS SoulsCoreTests SoulsCoreBench SoulsRuntime SoulsEditor SoulsSceneTests SoulsAllocationTests SoulsVulkanRuntimeCheck)
    if(TARGET ${app})
      add_custom_command(TARGET ${app} POST_BUILD
        COMMAND ${CMAKE_COMMAND} "-Dconfig=$<CONFIG>" "-Druntime=${SOULS_ASAN_RUNTIME_RELEASE}" "-Ddebug_runtime=${SOULS_ASAN_RUNTIME_DEBUG}" "-Ddestination=$<TARGET_FILE_DIR:${app}>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/DeployAsan.cmake")
    endif()
  endforeach()
endif()
